📌 레이아웃 패턴 1: 세로 배치 (기출 1번 유형)

구조: header → nav(가로 목록) → section > article들 (세로)

<!-- HTML -->
<header id="top">  <!-- id="top" : 나중에 상단으로 이동 앵커로 사용 -->
    <h1><a href="http://sae.kangnam.ac.kr" target="_blank">ICT공학부</a></h1>
</header>

<nav>
    <ul>
        <li><a href="#sec1">ICT공학부 소개</a></li>
        <li><a href="#sec2">교과과정</a></li>
        <li><a href="#sec3">진출분야</a></li>
    </ul>
</nav>

<section>
    <article id="sec1">
        <h2>ICT공학부 소개 <a href="#top"><img src="img/top.png"></a></h2>
        <!-- 내용... -->
        <hr>
    </article>

    <article id="sec2">
        <h2>교과과정 <a href="#top"><img src="img/top.png"></a></h2>
        <!-- 표 내용... -->
        <hr>
    </article>

    <article id="sec3">
        <h2>진출분야 <a href="#top"><img src="img/top.png"></a></h2>
        <!-- 내용... -->
        <hr>
    </article>
</section>

/* CSS - 레이아웃 패턴 1 */
* { margin: 0; padding: 0; box-sizing: border-box; }

body {
    font-size: 15px;          /* ★ 기본 폰트 크기 15px - 항상! */
    background-color: #f5f5e8; /* 배경색 */
    width: 100%;
}

/* 헤더 */
header {
    background-color: #003399;
    padding: 20px;
    text-align: center;
}
header h1 a {
    color: #0044ff;
    font-size: 30px;
    text-decoration: none;    /* ★ 링크 밑줄 제거 */
}

/* 내비게이션 - 가로 목록 */
nav { background-color: #e8e8d0; }
nav ul {
    list-style: none;         /* 목록 마커 제거 */
    display: flex;            /* 가로 배치 */
    padding: 5px;
}
nav ul li a {
    display: block;
    padding: 8px 20px;
    text-decoration: none;    /* ★ 링크 밑줄 제거 */
    color: black;
}
nav ul li a:hover {           /* ★ 마우스 올리면 배경색 변경 */
    background-color: #aaaaff;
}

/* article 제목: 인라인-블록, 20px, bold */
article h2 {
    display: inline-block;    /* ★ 인라인-블록 */
    font-size: 20px;          /* ★ 20픽셀 */
    font-weight: bold;        /* ★ bold */
    margin: 10px 0;
}

/* top.png 이미지 링크 */
article h2 a img {
    width: 20px;
    vertical-align: middle;
}
article h2 a { text-decoration: none; } /* ★ 이미지 링크 테두리 없애기 */

/* section 전체 */
section { padding: 10px 20px; }
article { margin-bottom: 20px; }

📌 레이아웃 패턴 2: nav 세로 + iframe 2개 (기출 2번 유형)

<!-- HTML -->
<header>
    <table>  <!-- header 내용은 table로 구현 -->
        <tr>
            <td><a href="http://kangnam.ac.kr" target="_blank">
                <img src="img/kangnam.gif"></a></td>
            <td><h1>Basic Web programming</h1></td>
        </tr>
    </table>
</header>

<nav>
    <ul>
        <li><a href="pc.html" target="left_frame">PC</a></li>  <!-- target으로 iframe 지정 -->
        <li><a href="http://etnews.co.kr" target="right_frame">신문사</a></li>
    </ul>
</nav>

<section>
    <article>
        <!-- name 속성으로 target과 연결 -->
        <iframe src="" name="left_frame" width="200px" height="600px"></iframe>
    </article>
    <article>
        <iframe src="" name="right_frame" width="800px" height="600px"></iframe>
    </article>
</section>

<footer>Copyright (c) 2020 웹개발기초, 홍길동</footer>

/* CSS - 레이아웃 패턴 2 */
body { font-size: 15px; }

header table { width: 100%; }
header img { text-decoration: none; border: none; } /* 이미지 링크 테두리 제거 */

nav {
    background-color: #333;   /* 진한 배경색 */
    width: 200px;             /* ★ 너비를 계산해서 pixel 값 설정 */
}
nav ul { list-style: none; padding: 0; }
nav ul li a {
    display: block;
    padding: 10px;
    color: white;
    text-decoration: none;
}
nav ul li a:hover { color: blue; } /* ★ 마우스 올리면 글자색 파란색 */

/* section의 두 article을 나란히 */
section { overflow: hidden; } /* float 해제용 */
section article {
    float: left;              /* ★ float: left로 나란히 */
    margin: 5px;
    border: 1px solid gray;
}

footer {
    background-color: lightgray; /* ★ footer 배경색 */
    font-size: 0.9em;            /* ★ 보통 글씨의 0.9배 */
    font-style: italic;          /* ★ 이탤릭체 */
    clear: both;
    padding: 5px;
}

/* ★ 링크 기본 스타일 제거 */
a { text-decoration: none; color: inherit; }

📌 레이아웃 패턴 3: nav 왼쪽 고정 + section 오른쪽

<!-- HTML -->
<div id="wrap">   <!-- 전체 너비 제한용 div -->
    <header>헤더 영역</header>

    <nav>
        <ul>
            <li><a href="#">메뉴1</a></li>
            <li><a href="#">메뉴2</a></li>
        </ul>
    </nav>

    <section>섹션 내용</section>

    <footer style="clear:both">푸터</footer>
</div>

/* CSS - 패턴 3: nav 왼쪽 + section 오른쪽 */
body { font-size: 15px; }

#wrap {
    width: 1000px;        /* 전체 너비 */
    margin: 0 auto;       /* ★ 브라우저 중앙 정렬 */
}

header { width: 100%; background-color: #003399; }

nav {
    width: 200px;
    float: left;          /* ★ nav를 왼쪽에 */
    background-color: #eee;
}

section {
    width: 800px;         /* 전체 - nav 너비 */
    float: left;          /* ★ section도 왼쪽으로 (nav 옆에 붙음) */
}

footer {
    clear: both;          /* ★ float 해제 */
    width: 100%;
}

nav ul { list-style: none; padding: 0; margin: 0; }
nav ul li a {
    display: block;
    padding: 10px 15px;
    text-decoration: none;
    color: black;
}
nav ul li a:hover {
    background-color: aliceblue; /* 마우스 올리면 배경색 */
    color: dodgerblue;
    cursor: pointer;             /* ★ 손 모양 커서 */
}
