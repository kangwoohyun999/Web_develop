📌 링크(a 태그) 패턴 총정리

<!-- ★ 새 창에서 열기 (target="_blank") -->
<a href="http://sae.kangnam.ac.kr" target="_blank">새 창에서 열기</a>

<!-- ★ 현재 창에서 열기 (기본값, target 생략 or "_self") -->
<a href="http://kangnam.ac.kr">현재 창</a>
<a href="http://kangnam.ac.kr" target="_self">현재 창</a>

<!-- ★ 페이지 내 특정 위치로 이동 (앵커) -->
<!-- 1단계: 이동할 목적지에 id 달기 -->
<header id="top">...</header>
<article id="sec1">...</article>

<!-- 2단계: #id로 링크 -->
<a href="#top">맨 위로</a>
<a href="#sec1">섹션1로 이동</a>

<!-- ★ 이미지 링크 (이미지를 클릭하면 이동) -->
<a href="http://kangnam.ac.kr" target="_blank">
    <img src="img/kangnam.gif" alt="강남대">
</a>

<!-- ★ 이미지 클릭하면 현재 문서로 이동 (href="#" 또는 href="파일명.html") -->
<a href="#">
    <img src="img/logo.png" width="200">
</a>

<!-- ★ top.png 클릭하면 맨 위로 이동 -->
<a href="#top"><img src="img/top.png" alt="top"></a>

<!-- 파일 다운로드 -->
<a href="file.pdf" download>다운로드</a>

/* ★ 링크 밑줄/색상 제거 - 항상 필수 */
a {
    text-decoration: none;  /* 밑줄 제거 */
    color: inherit;         /* 색상 부모에서 상속 */
}

/* 이미지 링크 테두리 제거 */
a img { border: none; }

/* 특정 링크만 스타일 */
nav a { color: black; text-decoration: none; }
nav a:hover { background-color: #aaa; color: blue; }

📌 iframe 활용 (기출 2번 유형)

<!-- ★ iframe 기본 -->
<iframe src="other.html" width="400" height="300"></iframe>

<!-- ★ 외부 사이트 삽입 -->
<iframe src="http://etnews.co.kr" width="800px" height="600px"></iframe>

<!-- ★ nav 링크 → 특정 iframe에 표시
     핵심: iframe에 name 지정, a 태그에 target=name -->

<!-- nav 메뉴 -->
<nav>
    <ul>
        <!-- target="left_frame" → name="left_frame" iframe에 표시 -->
        <li><a href="pc.html" target="left_frame">PC</a></li>
        <!-- target="right_frame" → name="right_frame" iframe에 표시 -->
        <li><a href="http://etnews.co.kr" target="right_frame">신문사</a></li>
    </ul>
</nav>

<section>
    <article>
        <!-- ★ name 속성으로 target과 연결 -->
        <iframe src="" name="left_frame" width="200px" height="600px"></iframe>
    </article>
    <article>
        <iframe src="" name="right_frame" width="800px" height="600px"></iframe>
    </article>
</section>

/* iframe이 있는 section article 나란히 배치 */
section article {
    float: left;         /* ★ 나란히 정렬 */
    border: 1px solid gray;
    margin: 5px;
}

/* iframe 자체 스타일 */
iframe {
    border: none;        /* 기본 테두리 제거 시 */
    display: block;
}

📌 페이지 내 앵커 이동 패턴 (기출 1번 B, C번)

<!-- 전체 구조 -->

<!-- 1. header에 id="top" 달기 (맨 위로 이동용) -->
<header id="top">
    <h1>ICT공학부</h1>
</header>

<!-- 2. nav 메뉴에서 #id로 링크 -->
<nav>
    <ul>
        <li><a href="#intro">ICT공학부 소개</a></li>
        <li><a href="#curriculum">교과과정</a></li>
        <li><a href="#career">진출분야</a></li>
    </ul>
</nav>

<section>
    <!-- 3. 각 article에 해당 id 달기 -->
    <article id="intro">
        <!-- ★ article 제목 오른쪽에 top.png 아이콘 -->
        <h2>ICT공학부 소개
            <a href="#top"><img src="img/top.png" alt="top"></a>
        </h2>
        <p>내용...</p>
        <hr>  <!-- ★ 각 영역 아래쪽에 수평선 -->
    </article>

    <article id="curriculum">
        <h2>교과과정
            <a href="#top"><img src="img/top.png" alt="top"></a>
        </h2>
        <!-- 표 내용 -->
        <hr>
    </article>

    <article id="career">
        <h2>진출분야
            <a href="#top"><img src="img/top.png" alt="top"></a>
        </h2>
        <p>내용...</p>
        <hr>
    </article>
</section>

📌 고정 TOP 버튼 (position: fixed)

<!-- HTML: body 어디든 넣기 -->
<a href="#top" id="top-btn">
    <img src="img/top.png" alt="TOP">
</a>
<!-- 또는 텍스트 버튼 -->
<a href="#top" id="top-btn">TOP</a>

/* ★ 오른쪽 하단 고정 TOP 버튼 */
#top-btn {
    position: fixed;     /* 스크롤해도 고정 */
    right: 30px;         /* 오른쪽에서 30px */
    bottom: 30px;        /* 아래에서 30px */
    text-decoration: none;
    cursor: pointer;
}
