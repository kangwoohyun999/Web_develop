📌 폼 입력 요소 총정리

<!-- ★ 텍스트 입력 -->
<input type="text" placeholder="영/숫자 6-10자" size="15">

<!-- ★ 비밀번호 -->
<input type="password" placeholder="영 대소문자/숫자 6-15자">

<!-- ★ 라디오 버튼 - name이 같으면 하나만 선택 -->
<input type="radio" name="gender" value="m"> 남성
<input type="radio" name="gender" value="f" checked> 여성  <!-- checked: 기본 선택 -->

<!-- ★ 체크박스 - 여러 개 선택 가능 -->
<input type="checkbox" name="agree" value="1"> 휴대전화
<input type="checkbox" name="agree" value="2" checked> 이메일

<!-- ★ 콤보박스(드롭다운) -->
<select name="phone">
    <option value="010" selected>010</option>
    <option value="017">017</option>
    <option value="019">019</option>
</select>

<!-- ★ 날짜 선택 -->
<input type="date" value="2022-04-01">

<!-- ★ 여러 줄 텍스트 -->
<textarea rows="5" cols="40" placeholder="내용 입력"></textarea>

<!-- ★ 버튼 -->
<button type="reset">처음으로</button>     <!-- 초기화 -->
<button type="submit">확인</button>       <!-- 전송 -->
<input type="button" value="닫기">        <!-- 일반 버튼 -->

<!-- fieldset으로 입력 그룹 묶기 -->
<fieldset>
    <legend>회원정보</legend>
    <input type="checkbox"> 개인정보 이용 동의(필수)<br>
    <input type="checkbox"> 서비스 이용 약관 동의(필수)<br>
    <input type="checkbox"> 제3자 정보제공 동의(선택)
</fieldset>

📌 회원가입 폼 - table로 정렬 (기출 mid-2번 유형)

<h2>회원가입 메뉴</h2>

<!-- ★ table을 이용하면 정렬하기 편리함 (힌트에서 매번 언급) -->
<table id="202001234">  <!-- ★ id에 학번 사용 -->
    <tr>
        <td>ID</td>
        <td><input type="text" placeholder="영/숫자 6-10자"></td>
    </tr>
    <tr>
        <td>비밀번호</td>
        <td><input type="password" placeholder="영 대소문자/숫자 6-15자"></td>
    </tr>
    <tr>
        <td>휴대전화</td>
        <td>
            <select name="phone">
                <option>010</option>
                <option>017</option>
                <option>019</option>
            </select>
            - <input type="text" placeholder="12345678">
        </td>
    </tr>
    <tr>
        <td>이메일</td>
        <td><input type="email" placeholder="id@domain"></td>
    </tr>
    <tr>
        <td>연락방법</td>
        <td>
            <input type="checkbox" name="contact" value="phone"> 휴대전화
            <input type="checkbox" name="contact" value="email"> 이메일
        </td>
    </tr>
    <tr>
        <td>개인정보</td>
        <td>
            <input type="radio" name="privacy" value="agree" checked> 동의
            <input type="radio" name="privacy" value="disagree"> 비동의
        </td>
    </tr>
    <tr>
        <td colspan="2" style="text-align:center">
            <button type="reset">처음으로</button>
            <button type="submit">확인</button>
        </td>
    </tr>
</table>

/* 회원가입 폼 CSS */

/* ★ 입력 상자에 마우스 올리면 테두리 빨간색 (5곳) */
input[type="text"]:hover,
input[type="password"]:hover,
input[type="email"]:hover,
input[type="tel"]:hover,
select:hover {
    border: 1px solid red;
}

/* ★ 입력 상자에 포커스(클릭)하면 배경색 변경 */
input:focus {
    background-color: lightyellow;
    outline: none;
}

/* ★ 버튼에 마우스 올리면 글씨 1.1배 */
button:hover {
    font-size: 1.1em;
    cursor: pointer;
}

/* 제출 버튼 스타일 */
button[type="submit"] {
    background-color: blue;
    color: white;
    padding: 8px 20px;
    border: none;
    cursor: pointer;
}

button[type="reset"] {
    background-color: #555;
    color: white;
    padding: 8px 20px;
    border: none;
    cursor: pointer;
}

/* 폼 테이블 정렬 */
table { border-collapse: collapse; }
table td { padding: 8px 12px; vertical-align: middle; }
table td:first-child { text-align: right; width: 100px; } /* 왼쪽 레이블 */

    </tr>
</table>

    outline: none;
