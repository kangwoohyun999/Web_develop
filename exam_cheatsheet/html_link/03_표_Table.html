📌 표(Table) 기본 구조

<table>
    <!-- colgroup: 컬럼 너비 일괄 설정 ★ -->
    <colgroup>
        <col style="width: 100px">   <!-- 1열: 100px -->
        <col style="width: 150px">   <!-- 2열: 150px -->
        <col style="width: 150px">   <!-- 3열: 150px -->
    </colgroup>

    <thead>  <!-- 표 머리 -->
        <tr>
            <th rowspan="2">학년</th>       <!-- rowspan: 아래로 2칸 합치기 -->
            <th colspan="2">학기</th>        <!-- colspan: 오른쪽으로 2칸 합치기 -->
        </tr>
        <tr>
            <th>1</th>
            <th>2</th>
        </tr>
    </thead>

    <tbody>  <!-- 표 본문 -->
        <tr>
            <td>3</td>
            <td>SW</td>
            <td>VR</td>
        </tr>
        <tr>
            <td>4</td>
            <td colspan="2">AI</td>  <!-- AI는 2칸 합치기 -->
        </tr>
    </tbody>
</table>

/* 표 CSS - 기출 4번 유형 (시맨틱 테이블) */

table {
    border-collapse: collapse;  /* ★ 필수: 셀 사이 간격 없애기 */
    width: 100%;                /* 또는 고정 너비 */
}

/* thead 스타일 - 배경색, 글씨 굵게 */
thead {
    background-color: steelblue;
    color: white;
    font-weight: bold;          /* ★ 글씨 진하게 */
}

/* thead의 th */
thead th {
    padding: 8px;
    text-align: center;
    border: 1px solid #ccc;
}

/* tbody의 td */
tbody td {
    padding: 8px;
    text-align: center;
    border: 1px solid #ccc;
    background-color: lightcyan; /* 본문 배경색 */
}

/* ★ 셀에 마우스 올리면 색 반전 + 커서 변경 */
tbody td:hover {
    background-color: #555;    /* 진한색으로 */
    color: white;
    cursor: pointer;           /* 손 모양 */
}

        </tr>

/* 교과과정 표 CSS */
table {
    border-collapse: collapse;
    width: 100%;
    margin: 10px 0;
}

thead tr { background-color: #888; color: white; }
thead th { padding: 8px; border: 1px solid #ccc; text-align: center; }

tbody td {
    padding: 8px;
    border: 1px solid #ccc;
    text-align: center;
    background-color: cyan;     /* 시험 결과 색상에 맞게 */
}

/* ★ 짝수/홀수 행 배경색 다르게 */
tbody tr:nth-child(even) { background-color: #e0ffff; }
tbody tr:nth-child(odd)  { background-color: cyan; }

📌 header를 table로 구현 (기출 2번 B번 유형)

<header>
    <table>
        <tr>
            <td style="width:30%">
                <!-- 클릭하면 새 창에서 kangnam.ac.kr 열기 -->
                <a href="http://kangnam.ac.kr" target="_blank">
                    <img src="img/kangnam.gif" alt="강남대">
                </a>
            </td>
            <td style="text-align:center">
                <h1>Basic Web programming</h1>
            </td>
        </tr>
    </table>
</header>

/* header table CSS */
header table { width: 100%; border-collapse: collapse; }
header table td { padding: 10px; vertical-align: middle; }
header img { border: none; }  /* ★ 이미지 테두리 제거 */
