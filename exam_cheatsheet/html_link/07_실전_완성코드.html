/* === style2.css (기출 2번용) === */

* { margin: 0; padding: 0; box-sizing: border-box; }
body { font-size: 15px; }
a { text-decoration: none; }
a img { border: none; }

header { background-color: #ddd; padding: 5px; }
header table { width: 100%; }
header table td { padding: 5px 10px; vertical-align: middle; }
header h1 { font-size: 22px; color: #333; }

/* nav: 진한 배경 + 가로 목록 */
nav {
    background-color: #555;   /* ★ 진한 배경색 */
    overflow: hidden;
}
nav ul { list-style: none; display: flex; }
nav ul li a {
    display: block;
    padding: 10px 30px;
    color: white;
    text-decoration: none;
}
nav ul li a:hover { color: blue; } /* ★ 마우스 올리면 파란색 */

/* section의 두 article 나란히 */
section { overflow: hidden; }
section article {
    float: left;            /* ★ float: left로 나란히 */
    margin: 10px;
    border: 1px solid #ccc;
}

/* footer */
footer {
    clear: both;
    background-color: lightgray;  /* ★ 배경색 */
    font-size: 0.9em;             /* ★ 0.9배 */
    font-style: italic;           /* ★ 이탤릭 */

    color: #555;
}
