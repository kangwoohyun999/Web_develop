핵심: isNew 플래그로 초기/이어서 구분, eval()로 수식 계산, C 클릭 시 "0"으로 초기화
$(function () {
  var expr  = "";
  var isNew = true;

  $(".btn-num, .btn-op").on("click", function () {
    var v = $(this).data("val");
    if (isNew) { expr = v; isNew = false; }
    else       { expr += v; }
    $("#display").val(expr);
  });

  $("#btnEq").on("click", function () {
    try {
      var result = eval(expr);
      $("#display").val(result);
      expr = String(result);
    } catch(e) { $("#display").val("오류"); }
  });

  $("#btnC").on("click", function () {
    expr = ""; isNew = true;
    $("#display").val("0");
  });
});

은행 계좌 시뮬레이션 (jQuery)

★ 이번 시험 유력 ▾
핵심: MSG 배열(7개), 라디오로 action 읽기, isNaN+음수 검증, 이자 = balance × 0.02
$(function () {
  var balance = 10000;
  var rate    = 0.02;
  var MSG = [
    "잔액 확인 완료", "저금 완료", "인출 완료", "이자 지급 완료",
    "저금 금액 오류", "인출 금액 오류", "인출 불가(잔액 부족)"
  ];

  $("#btnRun").on("click", function () {
    var action = $("input[name='action']:checked").val();
    var amt    = parseFloat($("#amount").val());
    $("#outInterest").val("");

    if (action === "check") {
      $("#outBalance").val(balance);  $("#msg").text(MSG[0]);

    } else if (action === "save") {
      if (isNaN(amt) || amt < 0) { $("#msg").text(MSG[4]); return; }
      balance += amt;
      $("#outBalance").val(balance);  $("#msg").text(MSG[1]);

    } else if (action === "withdraw") {
      if (isNaN(amt) || amt < 0) { $("#msg").text(MSG[5]); return; }
      if (amt > balance)          { $("#msg").text(MSG[6]); return; }
      balance -= amt;
      $("#outBalance").val(balance);  $("#msg").text(MSG[2]);

    } else if (action === "interest") {
      var interest = balance * rate;
      balance += interest;
      $("#outInterest").val(interest);
      $("#outBalance").val(balance);  $("#msg").text(MSG[3]);
    }
  });
});

성적처리 (jQuery)

▾
핵심: 등급/평점 함수 분리, toFixed(1), 평균평점 = Σ(학점×평점) / Σ학점
$(function () {
  var credits     = [3, 3];
  var attendScore = [0, -1, -2];  // 출석/지각/결석

  function getGrade(total) {
    if (total >= 90) return "A";
    if (total >= 80) return "B";
    if (total >= 70) return "C";
    if (total >= 60) return "D";
    return "F";
  }
  function getPoint(grade) {
    var map = { A:4.0, B:3.0, C:2.0, D:1.0, F:0.0 };
    return map[grade];
  }

  $("#btnCalc").on("click", function () {
    var totalCredit = 0, sumPoint = 0;
    for (var i = 0; i < 2; i++) {
      var attIdx = parseInt($("input[name='att"+i+"']:checked").val());
      var exam   = parseInt($("#exam"+i).val());
      var total  = exam + attendScore[attIdx];
      var grade  = getGrade(total);
      var point  = getPoint(grade);
      $("#total"+i).text(total);
      $("#grade"+i).text(grade);
      $("#point"+i).text(point.toFixed(1));
      sumPoint    += credits[i] * point;
      totalCredit += credits[i];
    }
    $("#avgPoint").text("평균평점: " + (sumPoint / totalCredit).toFixed(1));
  });
});

$(function () {

      alert("귀하의 ID는 'kangnam'입니다.");

      alert("실패! 다시 시도하세요.");

장바구니 시뮬레이션 (jQuery)

★ 이번 시험 유력 ▾
핵심: 체크박스 .each(), $("<li>")+span 2개+.append(), 구매→.show(), 완료→.empty()
$(function () {
  var total = 0;

  /* [추가] 버튼 */
  $("#btnAdd").on("click", function () {
    var checked = $("input[name='item']:checked");
    if (checked.length === 0) { alert("상품을 선택한 후 추가 버튼을 클릭하세요."); return; }
    checked.each(function () {
      var name  = $(this).data("name");
      var price = parseInt($(this).val());
      var $li   = $("<li>");
      $li.append($("<span>").text(name));
      $li.append($("<span>").text(price.toLocaleString() + "원"));
      $("#cartList").append($li);
      total += price;
      $(this).prop("checked", false);
    });
    $("#totalInput").val(total.toLocaleString());
    $("#msgArea").text("처리 메시지: 상품이 장바구니에 추가되었습니다.");
  });

  /* [구매] 버튼 */
  $("#btnBuy").on("click", function () {
    if ($("#cartList li").length === 0) { alert("상품을 추가한 후 구매 버튼을 클릭하세요."); return; }
    $("#payArea").show();
  });

  /* [구매 진행] 버튼 */
  $("#btnPayGo").on("click", function () {
    var pay = $("input[name='pay']:checked").val();
    if (!pay) { alert("결재 수단을 선택하세요."); return; }
    if (confirm(pay + "(으)로 진행하시겠습니까?")) {
      $("#payArea").hide();
      $("input[name='pay']").prop("checked", false);
      $("#cartList").empty();
      total = 0; $("#totalInput").val("0");
      $("#msgArea").text("처리 메시지: \"" + pay + "\"(으)로 결제완료.");
    }
  });

  /* [취소] 버튼 */
  $("#btnCancel").on("click", function () { $("#payArea").hide(); });
});

To-do List (jQuery)

▾
핵심: 라디오+콤보로 옵션 읽기, 클릭 시 할일→완료 이동(.clone()), 완료 클릭 시 .remove()
$(function () {
  function bindTodo($li) {
    $li.css("cursor","pointer").on("click", function () {
      var $done = $(this).clone();
      $(this).remove();
      $done.css("background","lightgreen").off("click").on("click", function(){ $(this).remove(); });
      $("#doneList").append($done);
    });
  }
  function resetOptions() {
    $("input[name='priority'][value='normal']").prop("checked", true);
    $("#urgency").val("보통");
  }
  $("#btnAdd").on("click", function () {
    var text = $("#todoInput").val().trim();
    if (text === "") { alert("할 일을 입력하세요."); return; }
    var priority = $("input[name='priority']:checked").val();
    var urgency  = $("#urgency").val();
    var $li = $("<li>").text("["+priority+"/"+urgency+"] "+text).css("background","#ddd");
    bindTodo($li); $("#todoList").append($li);
    $("#todoInput").val("").focus(); resetOptions();
  });
  $("#btnMenu").on("click", function () { $("#funcMenu").show(); });
  $("#btnClear").on("click", function () { $("#doneList").empty(); $("#funcMenu").hide(); });
  $("#btnMenuCancel").on("click", function () { $("#funcMenu").hide(); });
});

▾

공통

  • jQuery CDN <script> 삽입
  • 코드 전체 $(function(){}); 안에 작성

  • 오류 후 포커스: $("#inp").focus()

DOM 조작

  • 보이기/숨기기: .show() / .hide()
  • 동적 추가: .append() / .prepend()
  • 초기화: .empty()
  • 삭제: .remove()
