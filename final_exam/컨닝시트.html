  var price = parseInt($(this).val());

});

$("input[name='item']").each(function () {
  if (this.checked) { var v = this.value; }
});

/* 체크박스 개수 확인 */
if ($("input[name='item']:checked").length === 0) { alert("선택!"); }

▾

시험 직전 체크리스트

▾

공통

  • jQuery CDN <script> 삽입
  • 코드 전체 $(function(){}); 안에 작성
  • CSS / JS 모두 html 내부에 작성

폼 읽기

  • 라디오: $("input[name='...']:checked").val()
  • 체크박스: .each() 또는 :checked
  • 콤보박스: .val() 또는 option:selected

DOM 조작

  • 보이기/숨기기: .show() / .hide()
  • 동적 추가: .append() / .prepend()
  • 초기화: .empty()

검증

  • 빈 값: if (val === "")
  • 숫자: isNaN(amt) || amt < 0
  • 오류 후 포커스: $("#inp").focus()
  • 버튼 hover CSS 작성
