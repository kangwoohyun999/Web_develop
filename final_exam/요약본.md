### 2021-2 ① 계산기
**핵심 동작:** 숫자(0~4)/연산자(+, *) 버튼 클릭 → 수식 조합 → `=` 클릭 → 계산 결과 출력, `C` 클릭 → 0으로 초기화

```javascript
$(function () {
  var expr = "";           // 입력 중인 수식
  var isNew = true;        // 초기 상태 여부

  // 숫자/연산자 버튼 공통: data-val 속성에 값을 담아두는 방식
  $(".btn-num, .btn-op").on("click", function () {
    var v = $(this).data("val");   // "1","2","+","*" 등
    if (isNew) {
      expr = v;
      isNew = false;
    } else {
      expr += v;
    }
    $("#display").val(expr);
  });

  // = 버튼
  $("#btnEq").on("click", function () {
    try {
      var result = eval(expr);     // 수식 계산 (덧셈/곱셈만 허용)
      $("#display").val(result);
      expr = String(result);
    } catch(e) { $("#display").val("오류"); }
  });

  // C 버튼
  $("#btnC").on("click", function () {
    expr = "";
    isNew = true;
    $("#display").val("0");        // 0으로 초기화
  });
});
```
> **핵심 포인트:** 초기 상태(`isNew`)와 이어서 입력하는 상태 구분, `eval()` 로 계산

### 2022-1 ① 은행 계좌 시뮬레이션 ★ (이번 시험 유력)

```javascript
$(function () {
  var balance = 10000;
  var rate    = 0.02;
  var MSG = [
    "잔액 확인 완료","저금 완료","인출 완료","이자 지급 완료",
    "저금 금액 오류","인출 금액 오류","인출 불가(잔액 부족)"
  ];

  $("#btnRun").on("click", function () {
    var action = $("input[name='action']:checked").val();
    var amt    = parseFloat($("#amount").val());

    $("#outInterest").val("");

    if (action === "check") {
      $("#outBalance").val(balance);
      $("#msg").text(MSG[0]);

    } else if (action === "save") {
      if (isNaN(amt) || amt < 0) { $("#msg").text(MSG[4]); return; }
      balance += amt;
      $("#outBalance").val(balance);
      $("#msg").text(MSG[1]);

    } else if (action === "withdraw") {
      if (isNaN(amt) || amt < 0) { $("#msg").text(MSG[5]); return; }
      if (amt > balance)          { $("#msg").text(MSG[6]); return; }
      balance -= amt;
      $("#outBalance").val(balance);
      $("#msg").text(MSG[2]);

    } else if (action === "interest") {
      var interest = balance * rate;
      balance += interest;
      $("#outInterest").val(interest);
      $("#outBalance").val(balance);
      $("#msg").text(MSG[3]);
    }
  });
});
```

### 2022-1 ② 성적처리

```javascript
$(function () {
  var credits   = [3, 3];        // 학점 수 (상수 배열)
  var attendScore = [0, -1, -2]; // 출석/지각/결석

  function getGrade(total) {
    if (total >= 90) return "A";
    if (total >= 80) return "B";
    if (total >= 70) return "C";
    if (total >= 60) return "D";
    return "F";
  }
  function getPoint(grade) {
    var map = { A:4.0, B:3.0, C:2.0, D:1.0, F:0.0 };
    return map[grade];
  }

  $("#btnCalc").on("click", function () {
    var totalCredit = 0, sumPoint = 0;

    for (var i = 0; i < 2; i++) {
      var attVal = $("input[name='att" + i + "']:checked").val(); // 출결 값
      var attIdx = parseInt(attVal);                              // 0/1/2
      var exam   = parseInt($("#exam" + i).val());
      var total  = exam + attendScore[attIdx];
      var grade  = getGrade(total);
      var point  = getPoint(grade);

      $("#total" + i).text(total);
      $("#grade" + i).text(grade);
      $("#point" + i).text(point.toFixed(1));

      sumPoint    += credits[i] * point;
      totalCredit += credits[i];
    }
    var avg = (sumPoint / totalCredit).toFixed(1);
    $("#avgPoint").text("평균평점: " + avg);
  });
});
```

## 3. JavaScript ↔ jQuery 대조표

| 기능 | JavaScript | jQuery |
|------|-----------|--------|
| 요소 선택 | `document.getElementById("id")` | `$("#id")` |
| 텍스트 쓰기 | `el.textContent = "값"` | `$("#id").text("값")` |
| HTML 쓰기 | `el.innerHTML = "<b>...</b>"` | `$("#id").html("<b>...</b>")` |
| input 값 읽기 | `el.value` | `$("#id").val()` |
| input 값 쓰기 | `el.value = "값"` | `$("#id").val("값")` |
| 라디오 값 읽기 | `for r of radios: if(r.checked)` | `$("input[name]:checked").val()` |
| 체크박스 순회 | `cbs.forEach(cb => if cb.checked)` | `$("input:checked").each(fn)` |
| 숨기기 | `el.style.display = "none"` | `$("#id").hide()` |
| 보이기 | `el.style.display = "block"` | `$("#id").show()` |
| 태그 생성 | `document.createElement("li")` | `$("<li>")` |
| 자식 끝에 추가 | `parent.appendChild(child)` | `$("ul").append($li)` |
| 자식 앞에 추가 | `parent.insertBefore(child, first)` | `$("ul").prepend($li)` |
| 자식 전체 삭제 | `el.innerHTML = ""` | `$("ul").empty()` |
| 태그 자체 삭제 | `el.parentNode.removeChild(el)` | `$(el).remove()` |
| 이벤트 등록 | `el.addEventListener("click", fn)` | `$("#id").on("click", fn)` |
| 포커스 | `el.focus()` | `$("#id").focus()` |
| 비활성화 | `el.disabled = true` | `$("#id").prop("disabled", true)` |
| 스타일 변경 | `el.style.background = "blue"` | `$("#id").css("background","blue")` |
| 속성 변경 | `el.src = "new.png"` | `$("img").attr("src","new.png")` |
| oninput 이벤트 | `el.oninput = fn` | `$("#id").on("input", fn)` |

## 5. 시험 직전 최종 체크

**[공통]**
- [ ] jQuery CDN `<script>` 삽입
- [ ] 코드 전체 `$(function () { });` 안에 작성
- [ ] CSS / JS 모두 html 내부에 작성

**[폼 읽기]**
- [ ] 라디오: `$("input[name='...']:checked").val()`
- [ ] 체크박스: `.each()` 또는 `:checked` 필터
- [ ] 콤보박스: `.val()` 또는 `option:selected`

**[DOM 조작]**
- [ ] 텍스트 출력: `.text()` / `.val()` / `.html()`
- [ ] 보이기/숨기기: `.show()` / `.hide()`
- [ ] 동적 추가: `.append()` / `.prepend()`
- [ ] 초기화: `.empty()`
- [ ] 삭제: `.remove()`

**[검증]**
- [ ] 빈 값: `if (val === "")`
- [ ] 숫자: `if (isNaN(amt) || amt < 0)`
- [ ] 잔액 부족: `if (amt > balance)`
- [ ] 라디오 미선택: `if ($(...:checked).length < 1)`
- [ ] 오류 후 포커스: `$("#inp").focus()`

**[버튼]**
- [ ] hover CSS: `button:hover { background: blue; color: white; cursor: pointer; }`
